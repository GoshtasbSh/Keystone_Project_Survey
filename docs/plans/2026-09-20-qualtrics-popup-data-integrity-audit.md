# Qualtrics ↔ Dashboard Popup Data-Integrity Audit Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Prove — or disprove, with evidence — that every value the KeyStone dashboard shows in a survey point's popup is the exact value that respondent gave in the Qualtrics export, and make every popup question label byte-identical to the survey's own question text so a collaborator can Ctrl-F it in the CSV/Excel.

**Architecture:** Three independent sources of truth are compared, never assumed equal: (1) the **live dashboard** at `https://keystone-project-survey-blue.vercel.app/` driven through a real browser as admin, (2) the **raw Qualtrics CSVs** parsed directly by column-index and by `ImportId`, and (3) the **server code path** (`api/_processing.py` → `api/iaq-points.py` → `static/js/dashboard.js`) read line-by-line. A reusable ground-truth harness parses the CSVs once, then every audit table is generated by diffing harness output against captured live popup DOM. Discrepancies are fixed at the source (Python label constants / JS label rendering), then re-verified against the live deployment.

**Tech Stack:** Python 3 + pandas (CSV ground truth), pytest (regression tests for the label/index contracts), Playwright MCP (live dashboard drive-through + screenshots), the deployed Vercel serverless API (`/api/iaq-points`, `/api/survey-points`, `/api/analysis`) read with the admin's Supabase JWT.

**Step 2: Write the ground-truth harness**

`scripts/audit/qualtrics_ground_truth.py` must expose one function and a CLI. It deliberately **does not import** `api/_processing.py` — it is an independent reading of the CSV, so that agreement between the two means something.

```python
"""Independent Qualtrics CSV reader — the audit's ground truth.

Deliberately does NOT import api/_processing.py. If this module and the
production extractor agree, that agreement is evidence. If this module
imported the code under audit, agreement would be circular and worthless.
"""
from __future__ import annotations

import csv
import json
import re
from pathlib import Path


def load_survey(path: str | Path) -> dict:
    """Parse a 3-header-row Qualtrics CSV.

    Returns:
        {
          'n_cols': int,
          'short_names':   [str, ...],   # header row 0
          'question_text': [str, ...],   # header row 1 — VERBATIM survey text
          'import_ids':    [str|None],   # header row 2, QID per column
          'qid_to_idx':    {qid: idx},
          'rows': [ {'ResponseId': str, 'cells': [str, ...]}, ... ],
        }
    """
    path = Path(path)
    with path.open(newline='', encoding='utf-8-sig') as f:
        raw = list(csv.reader(f))
    if len(raw) < 3:
        raise ValueError(f"{path.name}: fewer than 3 header rows")

    short_names, question_text, meta_row = raw[0], raw[1], raw[2]
    if 'ImportId' not in ' '.join(meta_row):
        raise ValueError(f"{path.name}: row 3 is not ImportId metadata")

    import_ids: list[str | None] = []
    qid_to_idx: dict[str, int] = {}
    for idx, cell in enumerate(meta_row):
        m = re.search(r'"ImportId"\s*:\s*"([^"]+)"', cell or '')
        qid = m.group(1) if m else None
        import_ids.append(qid)
        if qid:
            qid_to_idx.setdefault(qid, idx)
            base = qid[:-5] if qid.endswith('_TEXT') else qid
            qid_to_idx.setdefault(base, idx)

    rid_idx = short_names.index('ResponseId')
    rows = [
        {'ResponseId': r[rid_idx], 'cells': r}
        for r in raw[3:]
        if len(r) > rid_idx and r[rid_idx].strip()
    ]
    return {
        'n_cols': len(short_names),
        'short_names': short_names,
        'question_text': question_text,
        'import_ids': import_ids,
        'qid_to_idx': qid_to_idx,
        'rows': rows,
    }


def value_by_qid(survey: dict, row: dict, qid: str) -> tuple[str, int | None]:
    """Return (cell_value, column_index) for a QID, ('', None) if absent."""
    idx = survey['qid_to_idx'].get(qid)
    if idx is None:
        base = qid[:-5] if qid.endswith('_TEXT') else qid
        idx = survey['qid_to_idx'].get(base)
    if idx is None or idx >= len(row['cells']):
        return '', None
    return row['cells'][idx].strip(), idx


if __name__ == '__main__':
    import sys
    s = load_survey(sys.argv[1])
    print(json.dumps({k: s[k] for k in ('n_cols', 'qid_to_idx')}, indent=2)[:2000])
    print(f"data rows: {len(s['rows'])}")
```

```python
"""Every popup label must be the survey's own question text, verbatim."""
import sys, pathlib
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[1] / 'api'))
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[1]))

import pytest
from _processing import SURVEY_QUESTIONS
from scripts.audit.qualtrics_ground_truth import load_survey

CSV = pathlib.Path(__file__).resolve().parents[1] / 'data' / \
    'Keystone Heights Survey - V1_April 15, 2026_13.25.csv'


@pytest.fixture(scope='module')
def survey():
    return load_survey(CSV)


def test_every_survey_question_label_is_verbatim_csv_text(survey):
    """canonical_text must appear verbatim in the CSV's question-text row.

    A paraphrase cannot be found with Excel's Find, which is the whole
    point of showing the question in the popup.
    """
    mismatches = []
    for field, meta in SURVEY_QUESTIONS.items():
        if len(meta) != 3:
            continue
        _idx, qid, canonical = meta
        col = survey['qid_to_idx'].get(
            qid if qid in survey['qid_to_idx']
            else (qid[:-5] if qid.endswith('_TEXT') else qid)
        )
        if col is None:
            mismatches.append((field, qid, 'QID NOT IN CSV', ''))
            continue
        actual = survey['question_text'][col].strip()
        if canonical.strip() != actual:
            mismatches.append((field, qid, canonical, actual))
    assert not mismatches, (
        f"{len(mismatches)} label(s) are not verbatim survey text:\n" +
        "\n".join(f"  {f} ({q})\n    code: {c!r}\n    csv:  {a!r}"
                  for f, q, c, a in mismatches)
    )
```

```python
"""Production extractor must agree with the independent CSV reader."""
import sys, pathlib
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[1] / 'api'))
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[1]))

import pytest
from _processing import _read_qualtric_csv, _extract_survey_extras, SURVEY_QUESTIONS
from scripts.audit.qualtrics_ground_truth import load_survey, value_by_qid

CSV = (pathlib.Path(__file__).resolve().parents[1] / 'data' /
       'Keystone Heights Survey - V1_April 15, 2026_13.25.csv')
N_POINTS = 10


def test_extractor_matches_independent_reader_for_10_responses():
    truth = load_survey(CSV)
    raw, qmap = _read_qualtric_csv(CSV.read_bytes())

    rid_col = 'ResponseId'
    finished = raw[raw['Finished'].astype(str).str.strip().str.lower().isin(['true', '1'])]
    finished = finished.reset_index(drop=True)
    assert len(finished) >= N_POINTS, f"only {len(finished)} completed responses"

    truth_by_rid = {r['ResponseId']: r for r in truth['rows']}
    diffs = []
    for i in range(N_POINTS):
        row = finished.iloc[i]
        rid = str(row[rid_col]).strip()
        t_row = truth_by_rid.get(rid)
        assert t_row is not None, f"{rid} missing from independent reader"

        extracted = _extract_survey_extras(row, qmap, df_columns=finished.columns)
        for field, meta in SURVEY_QUESTIONS.items():
            if len(meta) != 3:
                continue
            _idx, qid, _text = meta
            expected, col = value_by_qid(truth, t_row, qid)
            if col is None:
                continue  # QID absent from this export — covered by the label test
            got = str(extracted.get(field, '') or '').strip()
            # Recode translation is applied downstream of extraction, so compare
            # raw-to-raw: a numeric export's '6' is correct here, 'Like' is not.
            if got != expected.strip():
                diffs.append((rid, field, qid, col, expected, got))

    assert not diffs, (
        f"{len(diffs)} field(s) disagree with the independent reader:\n" +
        "\n".join(f"  {r} {f} ({q}) csv_col={c}: csv={e!r} extracted={g!r}"
                  for r, f, q, c, e, g in diffs)
    )
```
